function calcNeto(monto, metodo, lugar){
  const m = Number(monto)||0;
  let com = 0;
  if(metodo==='Carta') com = m * CARTA_COM;
  else if(metodo==='Link pagamento') com = (m * LINK_COM) + LINK_FISSO;
  let fijos = 0;
  if(lugar==='Studio B') fijos = STUDIO_B_EXTRA;
  else if(lugar==='Domicilio') fijos = DOMICILIO_EXTRA;
  const affitto = m * AFFITTO_RATE;
  const esEfectivo = (metodo==='Contanti');
  const base = Math.max(0, m - com - fijos - affitto);
  const iva = esEfectivo ? 0 : base * IVA_RATE;
  const neto = base - iva;
  return {bruto:m, com, fijos, affitto, base, iva, esEfectivo, neto};
}

// ---- CALENDARIO ----
function viewCalendario(){
  const {year,month,selectedDate} = state.cal;
  const hoy = hoyStr();
  const primerDia = new Date(year,month,1);
  const diasMes = new Date(year,month+1,0).getDate();
  let startDay = primerDia.getDay(); if(startDay===0) startDay=7;

  // Raccoglie SOLO le citas (array), senza duplicare da proxima_cita
  const citasMes = {};
  state.pacientes.forEach(p=>{
    (p.citas||[]).forEach(c=>{
      const d=c.fecha.split('-');
      if(parseInt(d[0])===year && parseInt(d[1])-1===month){
        if(!citasMes[c.fecha]) citasMes[c.fecha]=[];
        const isFutura = c.fecha >= hoy;
        citasMes[c.fecha].push({
          paciente:p.nombre, pacienteId:p.id,
          hora:c.hora||'', zona:c.zona||'',
          tipo: isFutura ? 'programmata' : 'svolta',
          nota:c.nota||''
        });
      }
    });
    // proxima_cita solo se non è già in citas
    if(p.proxima_cita){
      const d=p.proxima_cita.split('-');
      if(parseInt(d[0])===year && parseInt(d[1])-1===month){
        const hasCita = (p.citas||[]).some(c=>c.fecha===p.proxima_cita);
        if(!hasCita){
          if(!citasMes[p.proxima_cita]) citasMes[p.proxima_cita]=[];
          citasMes[p.proxima_cita].push({
            paciente:p.nombre, pacienteId:p.id,
            hora:p.hora_cita||'', zona:(p.zonas||[]).join(', '),
            tipo:'programmata', nota:''
          });
        }
      }
    }
  });

  let cells = '';
  for(let i=1;i<startDay;i++){
    const prevDay=new Date(year,month,1-i);
    cells=`<div class="cal-cell other-month">${prevDay.getDate()}</div>`+cells;
  }
  for(let d=1;d<=diasMes;d++){
    const dateStr=`${year}-${String(month+1).padStart(2,'0')}-${String(d).padStart(2,'0')}`;
    const citas=citasMes[dateStr]||[];
    const esHoy=dateStr===hoy;
    const esSel=dateStr===selectedDate;
    const hayCitas=citas.length>0;
    const esPasado=dateStr<hoy&&hayCitas;
    let cls='cal-cell';
    if(esHoy) cls+=' today';
    else if(esSel) cls+=' selected';
    else if(esPasado) cls+=' worked';
    else if(hayCitas) cls+=' has-citas';
    // Punti colorati: blu=programmata, verde=svolta
    const dots=citas.map(c=>`<div class="cal-dot" style="background:${c.tipo==='programmata'?'#3B82F6':'var(--ok)'}"></div>`).join('');
    cells+=`<div class="${cls}" onclick="selDia('${dateStr}')"><span>${d}</span>${hayCitas?`<div class="cal-dots">${dots}</div>`:''}</div>`;
  }

  const sel = selectedDate || hoy;
  const citasSel = (citasMes[sel]||[]).sort((a,b)=>(a.hora||'').localeCompare(b.hora||''));
  const esPasadoSel = sel < hoy;
  const esHoySel = sel === hoy;

  // Formato data leggibile
  const selDate = new Date(sel+'T12:00:00');
  const selLabel = `${GIORNI[selDate.getDay()]} ${selDate.getDate()} ${MESI[selDate.getMonth()]} ${selDate.getFullYear()}`;

  let panelHTML = `<div class="cal-panel-date">${esHoySel?'📅 Oggi — ':''}<span style="color:var(--verde)">${selLabel}</span></div>`;

  if(citasSel.length>0){
    panelHTML+=`<div style="font-size:12px;font-weight:700;color:var(--muted);margin-bottom:8px;text-transform:uppercase;letter-spacing:0.5px">Visite del giorno (${citasSel.length})</div>`;
    citasSel.forEach(c=>{
      const tipoClass = c.tipo;
      const tipoLabel = c.tipo==='svolta' ? '✅ Svolta' : '🔵 Programmata';
      panelHTML+=`<div class="cita-item ${tipoClass}" onclick="irA('paciente-ver',${c.pacienteId})">
        <div class="cita-hora ${tipoClass}">${c.hora||'Orario n/d'} — ${tipoLabel}</div>
        <div class="cita-pac">${esc(c.paciente)}</div>
        ${c.zona?`<div class="cita-sub">📍 ${esc(c.zona)}</div>`:''}
        ${c.nota?`<div class="cita-sub" style="margin-top:4px;font-style:italic">💬 ${esc(c.nota)}</div>`:''}
        <div style="font-size:11px;margin-top:5px;color:var(--muted)">Tocca per aprire la scheda →</div>
      </div>`;
    });
  } else {
    panelHTML+=`<div class="cal-panel-empty">📋 Nessuna visita in questo giorno</div>`;
  }

  if(!esPasadoSel){
    const horasOcupadas=citasSel.map(c=>c.hora).filter(Boolean);
    const horasLibres=HORAS_TRABAJO.filter(h=>!horasOcupadas.includes(h));
    if(horasLibres.length>0){
      panelHTML+=`<div style="font-size:12px;font-weight:700;color:var(--muted);margin:10px 0 6px;text-transform:uppercase">⚡ Orari disponibili</div>
      <div class="time-slots">
        ${horasLibres.map(h=>`<div class="free-slot"><span>🕐 ${h}</span><button class="btn btn-primary btn-sm" onclick="abrirNuevaCita('${sel}','${h}')">+ Visita</button></div>`).join('')}
      </div>`;
    }
    panelHTML+=`<button class="btn btn-primary" style="width:100%;justify-content:center;margin-top:10px" onclick="abrirNuevaCita('${sel}','')">+ Nuova Visita</button>`;
  } else {
    panelHTML+=`<button class="btn btn-secondary" style="width:100%;justify-content:center;margin-top:10px" onclick="abrirNuevaCita('${sel}','')">+ Aggiungi seduta storica</button>`;
  }

  return `
  <div style="display:flex;justify-content:space-between;align-items:center;margin-bottom:20px">
    <h2 style="font-size:20px;font-weight:800;font-family:'Fraunces',serif">Calendario Clinico</h2>
    <div style="display:flex;gap:8px">
      <button class="btn btn-secondary btn-sm" onclick="configurarHorario()">⚙ Configura Orario</button>
    </div>
  </div>
  <div class="cal-wrap">
    <div>
      <div class="card" style="padding:20px">
        <div class="cal-nav">
          <button class="cal-nav-btn" onclick="calNav(-1)">‹</button>
          <div class="cal-nav-title">${MESI[month]} ${year}</div>
          <button class="cal-nav-btn" onclick="calNav(1)">›</button>
        </div>
        <div class="cal-header-days">
          ${['Lu','Ma','Me','Gi','Ve','Sa','Do'].map(d=>`<div class="cal-day-name">${d}</div>`).join('')}
        </div>
        <div class="cal-grid">${cells}</div>
        <div class="legend" style="margin-top:12px">
          <div class="legend-item"><div class="legend-dot" style="background:#3B82F6"></div>Visita programmata</div>
          <div class="legend-item"><div class="legend-dot" style="background:var(--ok)"></div>Seduta svolta</div>
          <div class="legend-item"><div class="legend-dot" style="background:var(--bg)"></div>Nessuna visita</div>
        </div>
      </div>
    </div>
    <div class="card cal-panel" style="padding:20px;position:sticky;top:80px;align-self:start">
      <div style="font-size:15px;font-weight:700;margin-bottom:14px;color:var(--verde)">${esHoySel?'📅 OGGI':'📋 Dettaglio giorno'}</div>
      ${panelHTML}
    </div>
  </div>`;
}

function guardarNuevaCita(){
  const pacId=parseInt(document.getElementById('nc-pac').value);
  const p=state.pacientes.find(x=>x.id===pacId); if(!p) return;
  if(!p.citas) p.citas=[];
  const fecha=document.getElementById('nc-fecha').value;
  const hora=document.getElementById('nc-hora').value;
  const zona=document.getElementById('nc-zona').value;
  const nota=document.getElementById('nc-nota').value;
  // Valida data
  if(!fecha || isNaN(new Date(fecha).getTime())){ alert('Data non valida.'); return; }
  p.citas.push({id:Date.now(),fecha,hora,zona,nota});
  // Aggiorna sedute realizzate se visita passata
  if(fecha < hoyStr()){
    p.sesiones_realizadas = (p.sesiones_realizadas||0)+1;
    if(p.sesiones_restantes>0) p.sesiones_restantes--;
  }
  // Aggiorna proxima_cita
  aggiornaProssimaCita(p);
  guardar(); cerrarModal(); render(); toast('✅ Visita salvata');
}

function aggiornaProssimaCita(p){
  const hoy = hoyStr();
  const future = (p.citas||[]).filter(c=>c.fecha>=hoy).sort((a,b)=>a.fecha.localeCompare(b.fecha));
  if(future.length){
    p.proxima_cita = future[0].fecha;
    p.hora_cita = future[0].hora||'';
  } else {
    p.proxima_cita = '';
    p.hora_cita = '';
  }
}

function eliminarCita(pid,cid){
  if(!confirm('Eliminare questa visita?')) return;
  const p=state.pacientes.find(x=>x.id===pid); if(!p) return;
  p.citas=(p.citas||[]).filter(c=>c.id!==cid);
  aggiornaProssimaCita(p);
  guardar(); render(); toast('Visita eliminata');
}

function guardarChiusura(id){

function viewContabilidad(){
  const s=state.stats;
  let todosPagos=[];
  state.pacientes.forEach(p=>(p.pagos||[]).forEach(pg=>todosPagos.push({...pg,pacienteNombre:p.nombre,pacienteId:p.id})));
  todosPagos.sort((a,b)=>new Date(b.fecha)-new Date(a.fecha));

  // Filtri
  const fMes = state.contab_filtroMes||'';
  const fMet = state.contab_filtroMetodo||'';
  const fPac = state.contab_filtroPaciente||'';
  let pagiFiltrati = todosPagos.filter(pg=>{
    if(fMes && !(pg.fecha||'').startsWith(fMes)) return false;
    if(fMet && pg.metodo!==fMet) return false;
    if(fPac && !pg.pacienteNombre.toLowerCase().includes(fPac.toLowerCase())) return false;
    return true;
  });

  const porMetodo={};
  todosPagos.filter(pg=>pg.pagado).forEach(pg=>{ porMetodo[pg.metodo]=(porMetodo[pg.metodo]||0)+Number(pg.monto); });
  const maxVal=Math.max(...Object.values(porMetodo),1);

  const now=new Date();
  const mensili=[];
  for(let i=5;i>=0;i--){
    const d=new Date(now.getFullYear(), now.getMonth()-i, 1);
    const ms=`${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}`;
    let tot=0;
    state.pacientes.forEach(p=>(p.pagos||[]).forEach(pg=>{ if(pg.pagado&&(pg.fecha||'').startsWith(ms)) tot+=Number(pg.monto)||0; }));
    mensili.push({label:`${MESI_BREVE[d.getMonth()]} ${d.getFullYear()}`, total:tot, ms});
  }
  const maxMes=Math.max(...mensili.map(m=>m.total),1);
  const metodoIcons={'Contanti':'💵','Bonifico':'🏦','Carta':'💳','Satispay':'📱','Link pagamento':'🔗'};

  // Totali filtrati
  let totLordo=0,totNetto=0,totComm=0,totIva=0,totAffitto=0;
  pagiFiltrati.filter(pg=>pg.pagado).forEach(pg=>{
    const c=calcNeto(pg.monto,pg.metodo,pg.lugar);
    totLordo+=c.bruto; totNetto+=c.neto; totComm+=c.com; totIva+=c.iva; totAffitto+=c.affitto+c.fijos;
  });

  return `
  <h2 style="font-size:22px;font-weight:700;margin-bottom:20px;font-family:'Fraunces',serif">💶 Contabilità</h2>
  <div class="stats-grid" style="margin-bottom:20px">
    <div class="stat-card" style="border-top:3px solid var(--ok)"><div class="stat-icon">✅</div><div class="stat-label">Lordo Incassato</div><div class="stat-val">${s.bruto.toFixed(2)}€</div></div>
    <div class="stat-card" style="border-top:3px solid var(--verde)"><div class="stat-icon">💶</div><div class="stat-label">Netto Reale</div><div class="stat-val" style="color:var(--verde)">${s.neto.toFixed(2)}€</div><div class="stat-sub">dopo commissioni e IVA</div></div>
    <div class="stat-card" style="border-top:3px solid var(--warn)"><div class="stat-icon">⏳</div><div class="stat-label">Da Incassare (lordo)</div><div class="stat-val" style="color:var(--warn)">${s.pend.toFixed(2)}€</div></div>
    <div class="stat-card" style="border-top:3px solid var(--danger)"><div class="stat-icon">⚠️</div><div class="stat-label">Da Incassare (netto)</div><div class="stat-val" style="color:var(--danger)">${s.pendN.toFixed(2)}€</div></div>
  </div>
  <div style="display:grid;grid-template-columns:1fr 1fr;gap:20px;margin-bottom:20px">
    <div class="card" style="margin-bottom:0">
      <div class="card-title">📊 Incassi per metodo</div>
      <div class="chart-bar-wrap">
        ${Object.entries(porMetodo).sort((a,b)=>b[1]-a[1]).map(([m,v])=>`
          <div class="chart-bar-row">
            <div class="chart-bar-label">${(metodoIcons[m]||'💶')} ${m}</div>

          </div>

function buildCompareRows(activoVal, lugar){
  return [40,50,60,75,80,100].map(v=>{
    const cEfec = calcNeto(v,'Contanti',lugar||'Studio A');
    const cCarta = calcNeto(v,'Carta',lugar||'Studio A');
    const activo = Math.abs(v-activoVal)<0.01;
    return `<tr style="${activo?'background:var(--verdeL);':''}" onclick="calcUpdate(${v})">
      <td style="padding:8px;font-weight:${activo?'700':'500'};border-bottom:1px solid var(--border);cursor:pointer">${v}€</td>
      <td style="padding:8px;text-align:right;font-weight:700;color:var(--ok);border-bottom:1px solid var(--border)">${cEfec.neto.toFixed(2)}€</td>
      <td style="padding:8px;text-align:right;font-weight:600;color:var(--verde);border-bottom:1px solid var(--border)">${cCarta.neto.toFixed(2)}€</td>
    </tr>`;
  }).join('');
}

// Aggiornamento VELOCE della calcolatrice (senza render completo)
function calcUpdate(val){
  state.calcBruto = val;
  const input = document.getElementById('calc-input-main');
  if(input && input.value != val) input.value = val;
  const b = parseFloat(val)||0;
  const met = state.calcMetodo||'Carta';
  const lug = state.calcLugar||'Studio A';
  const c = calcNeto(b, met, lug);
  const pctTuyo = b>0 ? ((c.neto/b)*100) : 0;

  // Display principale
  const dispFinal = document.getElementById('calc-display-final');
  const dispSub = document.getElementById('calc-display-sub');
  if(dispFinal) dispFinal.textContent = b>0 ? c.neto.toFixed(2)+'€' : '—';
  if(dispSub) dispSub.textContent = b>0 ? `su ${b.toFixed(2)}€ lordi (${pctTuyo.toFixed(1)}% rimane a te)` : 'inserisci un importo per calcolare';

  // Breakdown
  const bd = document.getElementById('calc-breakdown-box');
  if(bd) bd.innerHTML = `
    <div class="calc-breakdown-row"><span class="cb-label">💶 Importo lordo</span><span class="cb-val">${b.toFixed(2)}€</span></div>
    ${c.com>0?`<div class="calc-breakdown-row"><span class="cb-label">💳 Commissione (${(CARTA_COM*100).toFixed(2)}%)</span><span class="cb-val cb-neg">-${c.com.toFixed(2)}€</span></div>`:''}

    <div class="calc-breakdown-row"><span class="cb-label">📊 Base imponibile</span><span class="cb-val">${c.base.toFixed(2)}€</span></div>
    ${c.esEfectivo?
      `<div class="calc-breakdown-row cb-zero"><span class="cb-label">🏛 IVA</span><span class="cb-val">0.00€ ✅ esente</span></div>`:
      `<div class="calc-breakdown-row"><span class="cb-label">🏛 IVA (35%)</span><span class="cb-val cb-neg">-${c.iva.toFixed(2)}€</span></div>`}

function exportarCSV(){
  const h=["Nome","Età","Telefono","Email","Zone","Stato","Sedute svolte","Sedute pacchetto","Prossima visita","Dolore iniziale","Dolore attuale","Motivo","Incassato lordo","Incassato netto","Da incassare lordo","Da incassare netto"];
  const rows=state.pacientes.map(p=>{
    let cb=0,cn=0,pb=0,pn=0;
    (p.pagos||[]).forEach(pg=>{
      const c=calcNeto(pg.monto,pg.metodo,pg.lugar);
      if(pg.pagado){cb+=c.bruto;cn+=c.neto;}else{pb+=c.bruto;pn+=c.neto;}
    });
    return [p.nombre,p.edad,p.telefono,p.email,(p.zonas||[]).join('|'),p.estado,p.sesiones_realizadas,p.sesiones_restantes,p.proxima_cita,p.dolor_inicial,p.dolor_actual,p.motivo,cb.toFixed(2),cn.toFixed(2),pb.toFixed(2),pn.toFixed(2)].map(v=>`"${(v||'').toString().replace(/"/g,'""')}"`);
  });
  const csv="data:text/csv;charset=utf-8,\uFEFF"+[h.join(','),...rows.map(r=>r.join(','))].join('\n');
  const a=document.createElement('a');
  a.href=encodeURI(csv); a.download=`FisioCRM_${hoyStr()}.csv`;
  document.body.appendChild(a); a.click(); document.body.removeChild(a);
  toast('✅ CSV esportato');
}

function exportarBackup(){
  const data = JSON.stringify({version:2, date:hoyStr(), pacientes:state.pacientes}, null, 2);
  const blob = new Blob([data], {type:'application/json'});
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href=url; a.download=`FisioCRM_Backup_${hoyStr()}.json`;
  document.body.appendChild(a); a.click(); document.body.removeChild(a);
  URL.revokeObjectURL(url);
  toast('✅ Backup esportato');
}

function importarBackup(){
  const input = document.createElement('input');
  input.type='file'; input.accept='.json';
  input.onchange = e=>{
    const file = e.target.files[0]; if(!file) return;
    const reader = new FileReader();
    reader.onload = ev=>{
      try{
        const data = JSON.parse(ev.target.result);
        const pacs = data.pacientes||data;
        if(!Array.isArray(pacs)) throw new Error('Formato non valido');
        if(!confirm(`Importare ${pacs.length} pazienti? I dati attuali verranno sostituiti.`)) return;
        state.pacientes = pacs;
        fixPacientes();
        guardar(); irA('dashboard'); toast(`✅ ${pacs.length} pazienti importati`);
      }catch(err){ alert('Errore nel file: '+err.message); }
    };
    reader.readAsText(file);
  };
  input.click();
}
